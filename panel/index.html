📸 صور شرح HWID لكل تطبيق

كل تطبيق يعرض معرّف الجهاز في مكان مختلف، فارفع له صور الشرح الخاصة به. تُرسل تلقائيًا للعميل مع رسالة طلب الـ HWID بعد اختياره للتطبيق — وإذا كانت أكثر من صورة تصله كألبوم واحد. يمكنك اختيار عدة صور دفعة واحدة (حتى 10 صور لكل تطبيق).

📖 صور شرح طريقة إضافة الملف داخل التطبيق

بعد وصول ملف الإعداد للعميل، تُرسل له تلقائيًا صور توضح كيف يستورد/يضيف الملف داخل التطبيق نفسه — بنفس آلية الألبوم (حتى 10 صور لكل تطبيق).

🖥️ عقد VPN (سيرفرات أخرى)

إن كان هذا البوت يعمل على سيرفر مستقل (مثل فرنسا) ويتحكم بسيرفرات VPN فعلية أخرى (مثل ألمانيا والسعودية) عبر الشبكة: سجّل كل سيرفر منها هنا كعقدة، ثم اربطها بمنطقة (بالأسفل) بنوع "عقدة VPN" ليختارها العميل. على كل سيرفر مستهدف نفسه، افتح لوحته وفعّله كعقدة من القسم أدناه، وانسخ التوكن والبصمة الظاهرين مرة واحدة إلى هنا.

Dart / Flutter API Client Implementation 
import 'dart:convert';
import 'package:http/http.dart' as http;

class FalconApiClient {
  final String baseUrl; // e.g. 'http://1.2.3.4:8080/mysecret'
  String? sessionCookie;

  FalconApiClient({required this.baseUrl});

  // 1. Authenticate & Obtain Session Cookie
  Future<bool> login(String username, String password) async {
    try {
      final response = await http.post(
        Uri.parse('$baseUrl/api/login'),
        headers: {'Content-Type': 'application/json'},
        body: jsonEncode({'username': username, 'password': password}),
      );
      if (response.statusCode == 200) {
        final rawCookie = response.headers['set-cookie'];
        if (rawCookie != null) {
          sessionCookie = rawCookie.split(';').first;
        }
        return true;
      }
    } catch (e) {
      print('Login error: $e');
    }
    return false;
  }

  // 2. Fetch Managed Users List
  Future<List<dynamic>> getUsers() async {
    final response = await http.get(
      Uri.parse('$baseUrl/api/users'),
      headers: {'Cookie': sessionCookie ?? ''},
    );
    if (response.statusCode == 200) {
      final data = jsonDecode(response.body);
      return data['users'] ?? [];
    }
    throw Exception('Failed to load users');
  }

  // 3. Create New User Account
  Future<bool> createUser({
    required String username,
    required String password,
    int days = 30,
    int connLimit = 1,
    double bandwidthGb = 0.0,
  }) async {
    final response = await http.post(
      Uri.parse('$baseUrl/api/users'),
      headers: {
        'Content-Type': 'application/json',
        'Cookie': sessionCookie ?? '',
      },
      body: jsonEncode({
        'username': username,
        'password': password,
        'days': days,
        'conn_limit': connLimit,
        'bandwidth_gb': bandwidthGb,
      }),
    );
    return response.statusCode == 200;
  }

  // 4. Lock or Unlock User
  Future<bool> toggleLock(String username, bool lock) async {
    final action = lock ? 'lock' : 'unlock';
    final response = await http.post(
      Uri.parse('$baseUrl/api/users/$username/$action'),
      headers: {'Cookie': sessionCookie ?? ''},
    );
    return response.statusCode == 200;
  }

  // 5. Renew Account Expiry
  Future<bool> renewUser(String username, int extendDays) async {
    final response = await http.post(
      Uri.parse('$baseUrl/api/users/$username/renew'),
      headers: {
        'Content-Type': 'application/json',
        'Cookie': sessionCookie ?? '',
      },
      body: jsonEncode({'days': extendDays}),
    );
    return response.statusCode == 200;
  }
}
Kotlin / OkHttp Android Client 
import okhttp3.*
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.RequestBody.Companion.toRequestBody
import org.json.JSONObject

class FalconApi(private val baseUrl: String) {
    private val client = OkHttpClient()
    private var sessionCookie: String? = null

    fun login(user: String, pass: String): Boolean {
        val json = JSONObject().put("username", user).put("password", pass).toString()
        val request = Request.Builder()
            .url("$baseUrl/api/login")
            .post(json.toRequestBody("application/json".toMediaType()))
            .build()

        client.newCall(request).execute().use { response ->
            if (response.isSuccessful) {
                sessionCookie = response.header("Set-Cookie")?.split(";")?.get(0)
                return true
            }
        }
        return false
    }

    fun getUsers(): String? {
        val request = Request.Builder()
            .url("$baseUrl/api/users")
            .addHeader("Cookie", sessionCookie ?: "")
            .get()
            .build()

        client.newCall(request).execute().use { response ->
            if (response.isSuccessful) {
                return response.body?.string()
            }
        }
        return null
    }
}
Swift / iOS URLSession Client 
import Foundation

class FalconApi {
    let baseUrl: String
    var sessionCookie: String?

    init(baseUrl: String) {
        self.baseUrl = baseUrl
    }

    func login(username: String, password: String, completion: @escaping (Bool) -> Void) {
        guard let url = URL(string: "\(baseUrl)/api/login") else { return }
        var request = URLRequest(url: url)
        request.httpMethod = "POST"
        request.addValue("application/json", forHTTPHeaderField: "Content-Type")
        let body = ["username": username, "password": password]
        request.httpBody = try? JSONSerialization.data(withJSONObject: body)

        URLSession.shared.dataTask(with: request) { _, response, error in
            if let httpResponse = response as? HTTPURLResponse, httpResponse.statusCode == 200 {
                if let cookies = HTTPCookieStorage.shared.cookies(for: url) {
                    for cookie in cookies where cookie.name == "session" {
                        self.sessionCookie = "\(cookie.name)=\(cookie.value)"
                    }
                }
                completion(true)
            } else {
                completion(false)
            }
        }.resume()
    }
}

🟢 سجل A (IPv4) & AAAA (IPv6)

يربط اسم النطاق بعنوان IP المباشر للسيرفر. استخدم A لعناوين IPv4 مثل 198.51.100.1 و AAAA لعناوين IPv6.

🔵 سجل CNAME (الاسم المستعار Alias)

يوجه نطاقاً فرعياً إلى نطاق آخر تلقائياً (مثل توجيه www.domain.com إلى domain.com أو لخدمات سحابية).

📄 سجل TXT (نصوص التحقق والبريد)

يستخدم لحفظ نصوص التحقق من ملكية النطاق (Google, Facebook, SSL) وسجلات أمان البريد مثل SPF و DKIM و DMARC لمنع التزوير.

✉️ سجل MX (خوادم البريد الإلكتروني)

يوجه رسائل البريد الإلكتروني الواردة إلى خادم البريد (مثل Google Workspace أو خادمك المحلي) مع تعيين رقم الأولوية (Priority).

🛡️ سجل CAA (تصريح سلطات الشهادات SSL)

يحدد الجهات المسموح لها حصراً بإصدار شهادات SSL لنطاقك (مثل letsencrypt.org) لمنع إصدار شهادات غير مصرح بها.

📡 سجلات NS & PTR & SRV

NS لتفويض خوادم أسماء فرعية، PTR لعنونة البحث العكسي، و SRV لتعريف مواقع الخدمات والمنافذ الخاصة (مثل SIP و Minecraft).

🟠 تشغيل البروكسي (Cloudflare Proxy - CDN)

يقوم بإخفاء الـ IP الحقيقي لسيرفرك خلف شبكة Cloudflare العالمية، مما يوفر حماية مجانية من هجمات الحجب (DDoS)، تشفير SSL تلقائي، وتخزين مؤقت للصفحات. يعمل مع اتصالات الويب (HTTP/HTTPS/WS).

⚪ الاتصال المباشر (DNS Only)

يوجه حركة المرور مباشرة إلى IP السيرفر دون وسيط Cloudflare. هذا الوضع إلزامي لخدمات البريد (MX) واتصالات الـ SSH المباشرة وبروتوكولات الـ VPN غير المدعومة عبر بروكسي Cloudflare.

🌐 ما هو الرمز @ والنطاقات الفرعية؟
  • @ (Root Domain): يمثل النطاق الرئيسي مباشرة (مثال: example.com). إذا تركت الحقل فارغاً سيعتمد النظام @ تلقائياً.
  • النطاق الفرعي (Subdomain): كتابة مثل vpn ستنشئ السجل تحت vpn.example.com.
  • الرمز * (Wildcard): يغطي كافة النطاقات الفرعية العشوائية تحت نطاقك.
⏱️ زمن بقاء الكاش (TTL) والأولوية (Priority)
  • TTL (Time To Live): المدة الزمنية التي تحتفظ بها خوادم الإنترنت بنسخة كاش من السجل. عند تفعيل البروكسي 🟠 تفرض Cloudflare خيار تلقائي (Auto) دائماً.
  • الأولوية (Priority): رقم للأفضلية في سجلات MX و SRV؛ الأرقام الأقل (مثل 10) تأخذ أولوية أولى قبل الأرقام الأعلى (مثل 20).
